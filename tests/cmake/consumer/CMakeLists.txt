cmake_minimum_required(VERSION 3.20)
project(RangeTapConsumerTest LANGUAGES C)
set(RNTP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${RANGETAP_SOURCE}" rangetap)
string(TOUPPER "${EXPECTED_BACKEND}" EXPECTED_BACKEND)
if(NOT RNTP_BACKEND STREQUAL EXPECTED_BACKEND)
    message(FATAL_ERROR "RNTP_BACKEND was not normalized: '${RNTP_BACKEND}'")
endif()
get_target_property(RNTP_TYPE rangetap TYPE)
if(NOT RNTP_TYPE STREQUAL "INTERFACE_LIBRARY")
    message(FATAL_ERROR "rangetap must remain header-only, got ${RNTP_TYPE}")
endif()
get_target_property(RNTP_DEFINITIONS rangetap INTERFACE_COMPILE_DEFINITIONS)
if(NOT "RNTP_BACKEND=RNTP_BACKEND_${EXPECTED_BACKEND}" IN_LIST RNTP_DEFINITIONS)
    message(FATAL_ERROR "Backend definition is not propagated: ${RNTP_DEFINITIONS}")
endif()
get_target_property(RNTP_INCLUDES rangetap INTERFACE_INCLUDE_DIRECTORIES)
if(NOT EXPECTED_BACKEND STREQUAL "NONE" AND NOT RNTP_MOCK_INCLUDE IN_LIST RNTP_INCLUDES)
    message(FATAL_ERROR "Backend include directory is not propagated: ${RNTP_INCLUDES}")
endif()
get_target_property(RNTP_LINKS rangetap INTERFACE_LINK_LIBRARIES)
if(EXPECTED_BACKEND STREQUAL "NONE" OR EXPECTED_BACKEND STREQUAL "NVTX")
    if(RNTP_LINKS)
        message(FATAL_ERROR "Header-only backend unexpectedly links libraries: ${RNTP_LINKS}")
    endif()
elseif(NOT RNTP_MOCK_LIBRARY IN_LIST RNTP_LINKS)
    message(FATAL_ERROR "Backend library is not propagated: ${RNTP_LINKS}")
endif()
add_library(consumer_a OBJECT main.c)
add_library(consumer_b OBJECT main.c)
target_link_libraries(consumer_a PRIVATE rangetap)
target_link_libraries(consumer_b PRIVATE rangetap)
