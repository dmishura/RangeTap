cmake_minimum_required(VERSION 3.20)
project(RangeTap VERSION 0.1.0 LANGUAGES C CXX)

set(RNTP_BACKEND "NONE" CACHE STRING "RangeTap profiler backend")
set_property(CACHE RNTP_BACKEND PROPERTY STRINGS NONE NVTX ITT STREAMLINE)
string(TOUPPER "${RNTP_BACKEND}" RNTP_BACKEND_NORMALIZED)
set(RNTP_VALID_BACKENDS NONE NVTX ITT STREAMLINE)
list(FIND RNTP_VALID_BACKENDS "${RNTP_BACKEND_NORMALIZED}" RNTP_BACKEND_INDEX)
if(RNTP_BACKEND_INDEX EQUAL -1)
    message(FATAL_ERROR "RangeTap: unsupported RNTP_BACKEND='${RNTP_BACKEND}'. Valid backends are: NONE, NVTX, ITT, STREAMLINE")
endif()
set(RNTP_BACKEND "${RNTP_BACKEND_NORMALIZED}" CACHE STRING "RangeTap profiler backend" FORCE)
set_property(CACHE RNTP_BACKEND PROPERTY STRINGS NONE NVTX ITT STREAMLINE)

add_library(rangetap INTERFACE)
add_library(RangeTap::RangeTap ALIAS rangetap)
target_include_directories(rangetap INTERFACE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>)
target_compile_features(rangetap INTERFACE c_std_99)
target_compile_definitions(rangetap INTERFACE RNTP_BACKEND=RNTP_BACKEND_${RNTP_BACKEND})

if(RNTP_BACKEND STREQUAL "NVTX")
    file(GLOB RNTP_NVTX_ROOT_HINTS LIST_DIRECTORIES true
        "/opt/nvidia/nsight-systems-cli/*/target-linux-x64/nvtx"
        "/opt/nvidia/nsight-systems/*/target-linux-x64/nvtx"
        "/opt/nvidia/nsight-compute/*/host/target-linux-x64/nvtx")
    find_path(RNTP_NVTX_INCLUDE_DIR NAMES nvtx3/nvToolsExt.h
        HINTS ${RNTP_NVTX_ROOT_HINTS} PATH_SUFFIXES include
        DOC "Directory containing the NVTX headers")
    if(NOT RNTP_NVTX_INCLUDE_DIR)
        message(FATAL_ERROR "RangeTap: NVTX headers not found; set RNTP_NVTX_INCLUDE_DIR")
    endif()
    target_include_directories(rangetap INTERFACE "${RNTP_NVTX_INCLUDE_DIR}")
elseif(RNTP_BACKEND STREQUAL "ITT")
    file(GLOB RNTP_ITT_ROOT_HINTS LIST_DIRECTORIES true "/opt/intel/oneapi/vtune/*")
    find_path(RNTP_ITT_INCLUDE_DIR NAMES ittnotify.h HINTS ${RNTP_ITT_ROOT_HINTS}
        PATH_SUFFIXES include sdk/include DOC "Directory containing the Intel ITT headers")
    find_library(RNTP_ITT_LIBRARY NAMES ittnotify HINTS ${RNTP_ITT_ROOT_HINTS}
        PATH_SUFFIXES lib64 sdk/lib64 DOC "Intel ITT static or shared library")
    if(NOT RNTP_ITT_INCLUDE_DIR OR NOT RNTP_ITT_LIBRARY)
        message(FATAL_ERROR "RangeTap: Intel ITT SDK not found; set RNTP_ITT_INCLUDE_DIR and RNTP_ITT_LIBRARY")
    endif()
    target_include_directories(rangetap INTERFACE "${RNTP_ITT_INCLUDE_DIR}")
    target_link_libraries(rangetap INTERFACE "${RNTP_ITT_LIBRARY}" "${CMAKE_DL_LIBS}")
elseif(RNTP_BACKEND STREQUAL "STREAMLINE")
    find_path(RNTP_STREAMLINE_INCLUDE_DIR NAMES streamline_annotate.h
        HINTS "$ENV{HOME}/gator/annotate" DOC "Directory containing the Arm Streamline annotation header")
    find_library(RNTP_STREAMLINE_LIBRARY NAMES streamline_annotate
        HINTS "$ENV{HOME}/gator/annotate" DOC "Arm Streamline annotation library")
    if(NOT RNTP_STREAMLINE_INCLUDE_DIR OR NOT RNTP_STREAMLINE_LIBRARY)
        message(FATAL_ERROR "RangeTap: Arm Streamline annotations not found; set RNTP_STREAMLINE_INCLUDE_DIR and RNTP_STREAMLINE_LIBRARY")
    endif()
    find_package(Threads REQUIRED)
    target_include_directories(rangetap INTERFACE "${RNTP_STREAMLINE_INCLUDE_DIR}")
    target_link_libraries(rangetap INTERFACE "${RNTP_STREAMLINE_LIBRARY}" Threads::Threads)
endif()

# Deprecated compatibility shim; backend selection is build-wide.
function(rangetap_link_backend target backend)
    if(NOT TARGET "${target}")
        message(FATAL_ERROR "RangeTap: target '${target}' does not exist")
    endif()
    string(TOUPPER "${backend}" backend)
    if(NOT backend STREQUAL RNTP_BACKEND)
        message(FATAL_ERROR "RangeTap: cannot select '${backend}' for '${target}'; this build uses RNTP_BACKEND=${RNTP_BACKEND}")
    endif()
    message(DEPRECATION "rangetap_link_backend() is deprecated; use target_link_libraries(${target} PRIVATE rangetap)")
    target_link_libraries("${target}" PRIVATE rangetap)
endfunction()

    set(RNTP_BUILD_TESTS_DEFAULT ON)
else()
    set(RNTP_BUILD_TESTS_DEFAULT OFF)
endif()
option(RNTP_BUILD_TESTS "Build RangeTap tests" ${RNTP_BUILD_TESTS_DEFAULT})
if(RNTP_BUILD_TESTS)
    enable_testing()
    if(RNTP_BACKEND STREQUAL "NONE")
        set(RNTP_COMPILE_TEST_SOURCE tests/compile/test_compile_none.c)
    elseif(RNTP_BACKEND STREQUAL "NVTX")
        set(RNTP_COMPILE_TEST_SOURCE tests/compile/test_compile_nvtx.c)
    elseif(RNTP_BACKEND STREQUAL "ITT")
        set(RNTP_COMPILE_TEST_SOURCE tests/compile/test_compile_itt.c)
    else()
        set(RNTP_COMPILE_TEST_SOURCE tests/compile/test_compile_streamline.c)
    endif()
    foreach(consumer IN ITEMS a b)
        add_executable(rntp_test_consumer_${consumer} "${RNTP_COMPILE_TEST_SOURCE}")
        target_link_libraries(rntp_test_consumer_${consumer} PRIVATE rangetap)
        add_test(NAME rntp_test_consumer_${consumer} COMMAND rntp_test_consumer_${consumer})
    endforeach()
    if(NOT RNTP_BACKEND STREQUAL "STREAMLINE")
        string(TOLOWER "${RNTP_BACKEND}" RNTP_BACKEND_LOWER)
        add_executable(rntp_runtime_${RNTP_BACKEND_LOWER} tests/runtime/test_runtime.c)
        target_link_libraries(rntp_runtime_${RNTP_BACKEND_LOWER} PRIVATE rangetap)
        add_test(NAME rntp_runtime_${RNTP_BACKEND_LOWER} COMMAND rntp_runtime_${RNTP_BACKEND_LOWER})
    endif()
    add_subdirectory(tests/cmake)
endif()
